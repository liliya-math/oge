/* ================= кэш ответов (ускорение чтения кабинета) =================
   Готовый ответ «кабинет ученика» и «список учеников» хранится в кэше скрипта до 2 минут.
   Любая запись (doPost, admin-операции кроме чтения, ручная правка таблицы) меняет «версию данных»,
   и старый кэш перестаёт использоваться. Если кэш недоступен или не влез, скрипт работает как раньше. */
const CACHE_TTL_ = 120;
const READ_OPS_ = ['ping', 'list', 'get', 'boardLibraryGet', 'quizList', 'quizGet', 'lessonDefaultsGet', 'lessonWindows', 'routeGet'];

function cacheVer_() {
  const c = CacheService.getScriptCache();
  let v = c.get('dv');
  if (!v) { v = String(Date.now()); c.put('dv', v, 21600); }
  return v;
}
function bumpVer_() {
  try { CacheService.getScriptCache().put('dv', Date.now() + Math.random().toString(36).slice(2, 6), 21600); } catch (e) {}
}
// Ручная правка таблицы тоже сбрасывает кэш (простой триггер, настраивать не нужно).
function onEdit() { bumpVer_(); }

function cachedRead_(key, fn) {
  const c = CacheService.getScriptCache();
  let k = null;
  try {
    k = 'c_' + cacheVer_() + '_' + key;
    const meta = c.get(k), n = Number(meta);
    if (meta && n > 0 && n <= 10) {
      const names = []; for (let i = 0; i < n; i++) names.push(k + '_' + i);
      const got = c.getAll(names);
      let b64 = '', ok = true;
      for (let i = 0; i < n; i++) { const part = got[names[i]]; if (part == null) { ok = false; break; } b64 += part; }
      if (ok) return JSON.parse(Utilities.ungzip(Utilities.newBlob(Utilities.base64Decode(b64), 'application/x-gzip')).getDataAsString());
    }
  } catch (err) { /* кэш не сработал: читаем из таблицы */ }
  const val = fn();
  if (k && val && !val.error) {
    try {
      const b64 = Utilities.base64Encode(Utilities.gzip(Utilities.newBlob(JSON.stringify(val), 'application/json')).getBytes());
      const SIZE = 90000, n = Math.ceil(b64.length / SIZE);
      if (n >= 1 && n <= 10) {
        const parts = {};
        for (let i = 0; i < n; i++) parts[k + '_' + i] = b64.slice(i * SIZE, (i + 1) * SIZE);
        c.putAll(parts, CACHE_TTL_);
        c.put(k, String(n), CACHE_TTL_);   // метка последней: читатель не увидит метку без частей
      }
    } catch (err) { /* не влезло: просто без кэша */ }
  }
  return val;
}

function setup() {
  const ss = SpreadsheetApp.getActive();
  Object.keys(SHEETS).forEach((name, i) => {
    let sh = ss.getSheetByName(name);
    if (!sh) sh = ss.insertSheet(name, i);
    const head = SHEETS[name];
    if (sh.getMaxColumns() < head.length) sh.insertColumnsAfter(sh.getMaxColumns(), head.length - sh.getMaxColumns());
    sh.getRange(1, 1, 1, head.length).setValues([head]);
    sh.setFrozenRows(1);
  });
  ['Sheet1', 'Лист1'].forEach(n => {
    const st = ss.getSheetByName(n);
    if (st && st.getLastRow() === 0 && ss.getSheets().length > 1) ss.deleteSheet(st);
  });
  const map = ss.getSheetByName('Карта');
  const rule = SpreadsheetApp.newDataValidation().requireValueInList(STATUSES, true).setAllowInvalid(false).build();
  map.getRange(2, 3, 300, 58).setDataValidation(rule);

  const hw = ss.getSheetByName('Домашка');
  hw.getRange(2, 7, 1000, 1).insertCheckboxes();
  hw.getRange(2, 4, 1000, 1).setNumberFormat('dd.MM.yyyy');
  hw.getRange(2, 8, 1000, 1).setNumberFormat('dd.MM.yyyy HH:mm');
  const hwSecRule = SpreadsheetApp.newDataValidation().requireValueInList(HW_SECS, true).setAllowInvalid(true).build();
  hw.getRange(2, 10, 1000, 1).setDataValidation(hwSecRule);
  ss.getSheetByName('Пробники').getRange(2, 2, 1000, 1).setNumberFormat('dd.MM.yyyy');
  ss.getSheetByName('Оценки').getRange(2, 2, 1000, 1).setNumberFormat('dd.MM.yyyy');
  const letters = ss.getSheetByName('Письма');
  letters.getRange(2, 4, 1000, 1).insertCheckboxes();
  letters.getRange(2, 2, 1000, 1).setNumberFormat('dd.MM.yyyy HH:mm');
  const sk = ss.getSheetByName('Навыки');
  sk.getRange(2, 3, 3000, 1).setDataValidation(rule);
  sk.getRange(2, 4, 3000, 1).setNumberFormat('dd.MM.yyyy');
  const us = ss.getSheetByName('Ученики');
  us.getRange(2, 12, 300, 1).insertCheckboxes();
  us.getRange(2, 6, 300, 1).setNumberFormat('@');
  us.getRange(2, 1, 300, 1).setNumberFormat('@');
  us.getRange(2, 18, 300, 1).setNumberFormat('@');

  if (us.getLastRow() >= 2) {
    const codes = us.getRange(2, 1, us.getLastRow() - 1, 1).getValues();

  }
  DriveApp.getRootFolder(); // запросить доступ к Диску сразу
  styleSheets();
  syncMaterials();
}

// Список для выпадающего меню берётся из catalog.json на сайте; если сайт недоступен, используется MATERIALS.
function materialsFromCatalog_() {
  try {
    const res = UrlFetchApp.fetch(CATALOG_URL, {muteHttpExceptions: true});
    if (res.getResponseCode() !== 200) return null;
    const items = (JSON.parse(res.getContentText()).items || []);
    const order = ['razbor', 'dz', 'trenazher', 'uchebnik', 'diagnostika', 'probnik', 'zadanie', 'sluzhebnoe', 'klass'];
    const rank = t => { const i = order.indexOf(t); return i < 0 ? order.length : i; };
    const out = items
      .filter(it => it && it.status === 'public' && String(it.title || '').trim() && /\.(html|pdf)$/i.test(String(it.file || '')) && it.type !== 'sistema')
      .sort((a, b) => rank(a.type) - rank(b.type))
      .map(it => [String(it.file), String(it.title).trim()]);
    return out.length > 20 ? out : null;
  } catch (e) { console.error('materialsFromCatalog_: ' + e); return null; }
}

  sh.getRange(2, 1, Math.max(sh.getMaxRows() - 1, 1), 2).clearContent();

/* ================= API для страницы ================= */
function doGet(e) {
  const p = (e && e.parameter) || {};
  const code = String(p.t || '').trim();
  if (p.schedule === 'capabilities') return json_({ok: true, lessonAvailability: 5});
  if (p.schedule === 'options') {
    const st = code && findStudentRow_(code);
    if (!st || st.row[11] === false) return json_({error: 'not_found'});
    return json_(lrPupilOptions_(SpreadsheetApp.getActive(), code, st, p.kind === 'weekly' ? 'weekly' : 'once', Date.now()));
  }
  if (p.quiz === 'capabilities') return json_({ok:true,quiz:1,ogeMocks:2});
  if (p.quiz === 'load') return json_(quizLoad_(code,{work:p.q,version:p.v,assignment:p.a,row:p.row}));
  if (p.w) return json_(dzLoad_(code, String(p.w).trim(), Number(p.row) || 0));
  return json_(code ? studentDataCached_(code, false) : {error: 'no_code'});
}

function doPost(e) {
  // Admin writes invalidate in admin_(). Reads must leave cached answers usable.
  let adminRequest = false;
  try {
    const body = JSON.parse(e.postData.contents);
    adminRequest = body.admin !== undefined;
  } catch (err) { /* Invalid requests are handled by doPostRun_(). */ }
  try { return doPostRun_(e); } finally { if (!adminRequest) bumpVer_(); }
}

function doPostRun_(e) {
  let body = {};
  try { body = JSON.parse(e.postData.contents); } catch (err) { return json_({error: 'bad_request'}); }
  if (body.admin !== undefined) return json_(admin_(body));
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    if (body.dz && String(body.t).trim() === 'demo') return json_(dzSubmit_('demo', body.dz));
    const st = findStudentRow_(body.t);
    if (!st || st.row[11] === false) return json_({error: 'not_found'});
    const code = String(body.t).trim();
    if (body.lessonRequest) {
      const result = lrRequest_(SpreadsheetApp.getActive(), code, st, body.lessonRequest, Date.now());
      return json_(result);
    }
    if (body.quiz) return json_(quizSubmit_(code, body.quiz));
    if (body.dz) return json_(dzSubmit_(code, body.dz));
    if (body.telegram === 'connect') return json_(telegramConnect_(code));
    if (body.workflow === true) return json_(workflowHw_(code, body, false));
    if (body.pet !== undefined) {
      const pet = String(body.pet);
      if (PETS.indexOf(pet) < 0) return json_({error: 'bad_value'});
      SpreadsheetApp.getActive().getSheetByName('Ученики').getRange(st.index, STUDENT_COLS.pet).setValue(pet);
      return json_({ok: true});
    }
    if (body.nick !== undefined) return json_(saveNick_(code, st, body.nick));
    if (body.letter !== undefined) return json_(saveLetter_(code, body.letter));
    return json_(setHwDone_(code, body.id, body.text, body.done));
  } finally { lock.releaseLock(); }
}

// Имя, по которому ученик просит к нему обращаться. Настоящие имя и фамилию видит только учитель.
function cleanNick_(v) {
  return String(v || '').replace(/[<>&"\\\u0000-\u001f]/g, '').replace(/\s+/g, ' ').trim().slice(0, 24);
}
function ensureNickCol_(sh, col) {
  col = col || STUDENT_COLS.nick;
  if (sh.getMaxColumns() < col) sh.insertColumnsAfter(sh.getMaxColumns(), col - sh.getMaxColumns());
  const h = sh.getRange(1, col);
  if (!String(h.getValue()).trim()) h.setValue(SHEETS['Ученики'][col - 1]);
}
function saveNick_(code, st, raw) {
  const nick = cleanNick_(raw);
  if (String(raw || '').trim() && !nick) return {error: 'bad_value'};
  const cache = CacheService.getScriptCache(), key = 'nick_' + code, cnt = Number(cache.get(key) || 0);
  if (cnt >= 10) return {error: 'too_many'};
  cache.put(key, String(cnt + 1), 3600);
  const sh = SpreadsheetApp.getActive().getSheetByName('Ученики');
  ensureNickCol_(sh);
  sh.getRange(st.index, STUDENT_COLS.nick).setValue(nick);
  return {ok: true, nickSaved: true, nick: nick, name: nick || String(st.row[1])};
}

function saveLetter_(code, text) {
  const t = String(text || '').trim().slice(0, 2000);
  if (t.length < 2) return {error: 'bad_value'};
  const cache = CacheService.getScriptCache(), key = 'letters_' + code, cnt = Number(cache.get(key) || 0);
  if (cnt >= 5) return {error: 'too_many'};
  cache.put(key, String(cnt + 1), 3600);
  let sh = SpreadsheetApp.getActive().getSheetByName('Письма');
  if (!sh) { sh = SpreadsheetApp.getActive().insertSheet('Письма', 99); sh.getRange(1, 1, 1, 4).setValues([SHEETS['Письма']]); sh.setFrozenRows(1); }
  sh.appendRow([code, new Date(), t, false]);
  notifyLetter_(code, t);
  return {ok: true};
}

// Письмо на почту владельца скрипта. Сбой отправки не мешает сохранению письма.
function notifyLetter_(code, text) {
  try {
    const st = findStudentRow_(code);
    const who = st ? String(st.row[1] || '').trim() : '';
    const to = Session.getEffectiveUser().getEmail();
    if (!to) return;
    MailApp.sendEmail({
      to: to,
      subject: 'Кабинет 25/25: новое письмо' + (who ? ' · ' + who : ''),

            '\n\nОткрыть кабинет в режиме учителя: ' + PAGE + '#lilia-' + code
    });
  } catch (e) { console.error('notifyLetter_: ' + e); }
}

function setHwDone_(code, id, text, done) {
  const sh = SpreadsheetApp.getActive().getSheetByName('Домашка');
  const row = Number(id);
  if (!(row >= 2 && row <= sh.getLastRow())) return {error: 'no_task'};
  const v = sh.getRange(row, 1, 1, 2).getValues()[0];
  if (String(v[0]).trim() !== code || String(v[1]).trim() !== String(text || '').trim()) return {error: 'changed'};
  sh.getRange(row, 7, 1, 2).setValues([[!!done, done ? new Date() : '']]);
  if (sh.getMaxColumns() >= 16) { sh.getRange(row, 11).setValue(done ? 'submitted' : 'new'); sh.getRange(row, 16).setValue(Utilities.getUuid()); }
  return {ok: true};
}

/* ================= режим учителя ================= */
function admin_(b) {
  const isRead = READ_OPS_.indexOf(b && b.op) >= 0;
  let r;
  try { r = adminRun_(b); } catch (err) { if (!isRead) bumpVer_(); throw err; }
  if (!isRead && !(r && (r.error === 'bad_password' || r.error === 'locked'))) bumpVer_();
  return r;
}

function adminRun_(b) {
  const pass = PropertiesService.getScriptProperties().getProperty('ADMIN');
  if (!pass) return {error: 'no_password'};
  const props = PropertiesService.getScriptProperties();
  const guard = LockService.getScriptLock();
  guard.waitLock(10000);
  let denied = null;
  try {
    const now = Date.now(), until = Number(props.getProperty('ADM_LOCK_UNTIL') || 0);
    if (until > now) denied = {error: 'locked', minutes: Math.ceil((until - now) / 60000)};
    else if (String(b.admin || '') !== pass) {
      const fails = Number(props.getProperty('ADM_FAILS') || 0) + 1;
      if (fails >= 5) { props.setProperty('ADM_LOCK_UNTIL', String(now + 15 * 60000)); props.setProperty('ADM_FAILS', '0'); }
      else props.setProperty('ADM_FAILS', String(fails));
      denied = {error: 'bad_password'};
    } else if (props.getProperty('ADM_FAILS') !== '0') props.setProperty('ADM_FAILS', '0');
  } finally { guard.releaseLock(); }
  if (denied) { Utilities.sleep(700); return denied; }
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = SpreadsheetApp.getActive();
    const code = String(b.t || '').trim();
    const need = () => { const st = findStudentRow_(code); if (!st) throw new Error('not_found'); return st; };
    const fresh = () => ({ok: true, d: studentData_(code, true)});
    switch (b.op) {
      case 'ping': return {ok: true, hwEdit: true, workflow: 1, quiz: 1, boardLibrary: 1, plan: 1, lessonRoute: 1};
      case 'routeGet': return {ok: true, materials: routeMaterials_(b.id)};
      case 'routeSave': return routeSave_(b);
      case 'routeUpload': return routeUpload_(b);
      case 'routeAssign': { const st = need(); return routeAssign_(ss, st, code, b); }
      case 'boardLibraryGet': return {ok:true,library:boardLibraryGet_()};
      case 'boardLibraryPut': return boardLibraryPut_(b);
      case 'quizList': return quizList_();
      case 'quizGet': { const d=quizDefinition_(String(b.work||''),Number(b.version)||0); return {ok:true,work:d,ogePart1:!!d&&quizMockFlag_(d.id,d.version,d.section)}; }
      case 'quizClassify': return quizClassify_(b);
      case 'quizSave': return quizSave_(b);
      case 'quizAssign': { need(); return quizAssign_(code,b); }
      case 'hwReview': { need(); const result = workflowHw_(code, b, true); return result.ok ? fresh() : result; }
      case 'lessonDefaultsGet': return {ok: true, minutes: lrDefaults_(), availability: lrAvailability_()};
      case 'lessonAvailabilitySet': {
        const availability = lrValidateAvailability_({extra: lrAvailability_().extra, ...b.availability});
        props.setProperty('LESSON_AVAILABILITY', JSON.stringify(availability));
        return {ok: true};
      }
      case 'lessonWindows': {
        if (need().row[11] === false) return {error: 'inactive'};
        const availability = lrAvailability_();
        const now = Date.now(), day = lrSamaraDay_(now), end = day + LR_TRANSFER_DAYS * 86400000;
        const busy = lrBusy_(ss, day - LR_BREAK_LOOKAROUND_MS, end + LR_BREAK_LOOKAROUND_MS, code, b.from);
        return {ok: true, lessonAvailability: 5, configured: availability.week.some(Boolean) || !!availability.extra?.length,
          windows: lrWindows_(availability, busy, now, end, b.expected)};
      }
      case 'lessonDefaultsSet': {
        props.setProperty('LESSON_DEFAULTS', JSON.stringify(lrMinutes_(b.minutes)));
        return {ok: true, minutes: lrDefaults_()};
      }
      case 'lessonReminderSet': {
        need();
        const cfg = lrConfig_(code);
        cfg.minutes = b.minutes === null ? null : lrMinutes_(b.minutes);
        lrSave_(code, cfg);
        return fresh();
      }
      case 'lessonRequestDecision': {
        const st = need();
        const result = lrDecideRequest_(ss, code, st, b, Date.now());
        return result.error ? result : fresh();
      }
      case 'lessonMove': {
        const st = need(), cfg = lrConfig_(code), now = Date.now();
        if (st.row[11] === false) return {error: 'inactive'};
        const from = b.from, to = b.to;
        const previous = cfg.moves.find(m => m.from === from);
        if (!Number.isSafeInteger(from) || (previous ? previous.to <= now : from <= now) || (to === null && from <= now) || (to !== null &&
          (!Number.isSafeInteger(to) || to <= now || (to - lrSamaraDay_(to)) % LR_LESSON_MS || to >= lrSamaraDay_(now) + LR_TRANSFER_DAYS * 86400000))) return {error: 'bad_value'};
        const slots = lrParse_(st.row[4], st.time, 'Europe/Samara').slots;
        if (!previous && !lrUpcoming_(slots, now, 8).includes(from)) return {error: 'bad_value'};
        if ((previous ? previous.to : from) !== b.expected) return {error: 'changed'};
        const target = to === null ? from : to;
        if (!lrFits_(target, lrAvailability_())) return {error: 'lesson_unavailable'};
        const day = lrSamaraDay_(target);
        const busy = lrBusy_(ss, day - LR_BREAK_LOOKAROUND_MS, day + 86400000 + LR_BREAK_LOOKAROUND_MS, code, from);
        const error = lrMoveError_(target, busy);
        if (error) return {error};
        cfg.moves = cfg.moves.filter(m => m.from !== from);
        if (to !== null && to !== from) cfg.moves.push({from, to});
        lrSave_(code, cfg);
        return fresh();
      }
      case 'list': return {ok: true, students: listStudentsCached_()};
      case 'rotateCode': { need(); return {ok: true, code: rotateCode_(code)}; }
      case 'get': { const d = studentDataCached_(code, true); return d.error ? {error: d.error} : {ok: true, d: d}; }
      case 'addStudent': {
        const name = String(b.name || '').trim();
        if (!name) return {error: 'no_name'};
        return {ok: true, code: createStudent_(name, String(b.dat || '').trim(), b.prog, Number(b.cls) || undefined)};
      }
      case 'student': {
        const st = need(), sh = ss.getSheetByName('Ученики'), f = b.fields || {};
        const pr = progOf_(st.row, f);
        if (f.nick !== undefined || f.routes !== undefined) ensureNickCol_(sh, f.routes !== undefined ? STUDENT_COLS.routes : STUDENT_COLS.nick);
        Object.keys(f).forEach(k => {
          if (!STUDENT_COLS[k]) return;
          let v = f[k];
          if (k === 'work') v = (Array.isArray(v) ? v : String(v).split('\n')).map(s => String(s).trim()).filter(Boolean).join('; ');
          else if (k === 'prog') v = normProg_(v);
          else if (k === 'goal' || k === 'lessonsPlan' || k === 'lessonsDone') v = Math.max(0, Math.min(k === 'goal' ? goalMax_(pr) : 99, Number(v) || 0));
          else if (k === 'active') v = !!v;
          else if (k === 'bdate') v = /^\d{4}-\d{2}-\d{2}$/.test(String(v || '')) ? new Date(String(v) + 'T00:00:00') : '';
          else if (k === 'cls') v = Math.max(0, Math.min(11, Number(v) || 0));
          else if (k === 'show') v = (Array.isArray(v) ? v : String(v).split(',')).map(x => String(x).trim()).filter(x => SHOW_KEYS.indexOf(x) >= 0).join(',');
          else if (k === 'pet') v = PETS.indexOf(String(v)) >= 0 ? String(v) : '';
          else if (k === 'nick') v = cleanNick_(v);
          else if (k === 'routes') v = (Array.isArray(v) ? v : String(v).split(',')).map(x => String(x).trim()).filter((x, i, a) => ROUTE_PROGS.indexOf(x) >= 0 && a.indexOf(x) === i).join(',');
          else v = String(v).trim();
          sh.getRange(st.index, STUDENT_COLS[k]).setValue(v);
        });
        if (f.name) { const m = findRow_('Карта', r => String(r[0]).trim() === code); if (m) ss.getSheetByName('Карта').getRange(m, 2).setValue(String(f.name).trim()); }
        return fresh();
      }
      case 'map': {
        const st = need();
        const n = Number(b.n), s = STATUS_RU[b.s];
        if (!(n >= 1 && n <= nMax_(progOf_(st.row))) || !s) return {error: 'bad_value'};
        const sh = ss.getSheetByName('Карта');
        let row = findRow_('Карта', r => String(r[0]).trim() === code);
        if (!row) { sh.appendRow([code, st.row[1]].concat(Array(58).fill('не проходили'))); row = sh.getLastRow(); }
        sh.getRange(row, n + 2).setValue(s);
        return fresh();
      }
      case 'skill': {
        need();
        const id = String(b.n || '').trim().toUpperCase(), s = STATUS_RU[b.s];
        if (!SKILL_RE.test(id) || !s) return {error: 'bad_value'};
        const sh = sheet_(ss, 'Навыки');
        const row = findRow_('Навыки', r => String(r[0]).trim() === code && String(r[1]).trim().toUpperCase() === id);
        if (b.s === 'new') { if (row) sh.deleteRow(row); }   // «не проверяли» = строки в таблице нет
        else if (row) sh.getRange(row, 3, 1, 2).setValues([[s, new Date()]]);
        else sh.appendRow([code, id, s, new Date()]);
        return fresh();
      }
      case 'planSave': {
        need();
        const res = planSave_(code, b.text);
        return res.error ? res : fresh();

/* ================= план занятий =================
   Формат текста (одна строка = один урок), поля через «|»:
     2026-11-05 | Алгебра | Тема урока | Домашнее задание | пометка
   Обязательны дата (ГГГГ-ММ-ДД), блок и тема; ДЗ и пометка необязательны.
   Пустые строки и строки, начинающиеся с #, пропускаются. */
const PLAN_MAX_ROWS = 300, PLAN_MAX_CHARS = 45000;
function planParse_(text) {
  const rows = [];
  const lines = String(text || '').replace(/\r\n?/g, '\n').split('\n');
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    if (!line || line.charAt(0) === '#') continue;
    const f = line.split('|').map(s => s.trim());
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(f[0] || '');
    const dt = m ? new Date(Date.UTC(+m[1], +m[2] - 1, +m[3])) : null;
    const valid = dt && dt.getUTCFullYear() === +m[1] && dt.getUTCMonth() === +m[2] - 1 && dt.getUTCDate() === +m[3];
    if (!valid || f.length < 3 || !f[1] || !f[2]) return {error: 'plan_bad_line_' + (i + 1)};
    rows.push({d: f[0], b: f[1].slice(0, 40), t: f[2].slice(0, 400), hw: (f[3] || '').slice(0, 600), n: (f[4] || '').slice(0, 120)});
  }
  rows.sort((a, b) => a.d < b.d ? -1 : a.d > b.d ? 1 : 0);
  return {rows: rows};
}
function planGet_(code) {
  const row = findRow_('Планы', r => String(r[0]).trim() === code);
  if (!row) return {text: '', rows: []};
  const text = String(SpreadsheetApp.getActive().getSheetByName('Планы').getRange(row, 2).getValue() || '');
  const p = planParse_(text);
  return {text: text, rows: p.rows || []};
}
function planSave_(code, text) {
  text = String(text || '').replace(/\r\n?/g, '\n').trim();
  if (text.length > PLAN_MAX_CHARS) return {error: 'plan_too_long'};
  const p = planParse_(text);
  if (p.error) return p;
  if (p.rows.length > PLAN_MAX_ROWS) return {error: 'plan_too_long'};
  const sh = SpreadsheetApp.getActive().getSheetByName('Планы');
  if (!sh) return {error: 'plan_no_sheet'};
  const row = findRow_('Планы', r => String(r[0]).trim() === code);
  if (!text) { if (row) sh.deleteRow(row); return {ok: true}; }
  if (row) sh.getRange(row, 2, 1, 2).setValues([[text, new Date()]]);
  else sh.appendRow([code, text, new Date()]);
  return {ok: true};
}

// Меняет код (ссылку) ученика во всех листах. Старая ссылка перестаёт работать сразу.
function rotateCode_(old) {
  const ss = SpreadsheetApp.getActive(), next = newCode_();
  const swap = (sh, col) => {
    if (!sh || sh.getLastRow() < 2) return;
    const rng = sh.getRange(2, col, sh.getLastRow() - 1, 1), v = rng.getValues();
    let hit = false;
    v.forEach(r => { if (String(r[0]).trim() === old) { r[0] = next; hit = true; } });
    if (hit) rng.setValues(v);
  };
  ['Ученики', 'Карта', 'Домашка', 'Пробники', 'Оценки', 'Письма', 'Навыки', 'Комментарии', 'Файлы', 'Разборы', 'Результаты', 'Попытки тестов', 'Занятия по карте']
    .forEach(n => swap(ss.getSheetByName(n), 1));
  const st = findStudentRow_(next);
  if (st) ss.getSheetByName('Ученики').getRange(st.index, 13).setValue(PAGE + '#' + next);
  const tg = telegramSS_();
  swap(tg.getSheetByName('Telegram'), 1);
  swap(tg.getSheetByName('Telegram-ссылки'), 2);
  swap(tg.getSheetByName('Telegram-файлы'), 2);
  const props = PropertiesService.getScriptProperties(), config = props.getProperty('LESSON_' + old);
  if (config) {
    props.setProperty('LESSON_' + next, config);
    props.deleteProperty('LESSON_' + old);
  }
  const log = ss.getSheetByName('Напоминания');
  if (log && log.getLastRow() > 1) {
    const range = log.getRange(2, 1, log.getLastRow() - 1, 1), keys = range.getValues();
    keys.forEach(row => { if (String(row[0]).startsWith(old + '|')) row[0] = next + String(row[0]).slice(old.length); });
    range.setValues(keys);
  }
  return next;
}
/* ================= Простая автопроверка домашки =================
 * Условия и ответы лежат в листе «Ключи» (по строке на задание), страница dz.html показывает только условия.
 * Ответ ученика сравнивается на сервере: ключи ученику не видны. Результат — в листе «Результаты» и на почту учителю. */
function dzRows_(work) {
  const sh = sheet_(SpreadsheetApp.getActive(), 'Ключи');
  if (!sh || sh.getLastRow() < 2) return [];
  return sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues()
    .filter(r => String(r[0]).trim() === work && String(r[2]).trim() !== '')
    .map(r => ({title: String(r[1] || '').trim(), n: String(r[2]).trim(), lv: String(r[3] || '').trim(), q: String(r[4] || ''), a: String(r[5] == null ? '' : r[5])}));
}
function dzNorm_(s) { return String(s == null ? '' : s).replace(/[−–—]/g, '-').replace(/\s+/g, '').replace(/,/g, '.').toLowerCase(); }
function dzSame_(given, key) {
  const g = dzNorm_(given);
  if (!g) return false;
  return String(key).split('|').some(k => {
    const kk = dzNorm_(k);
    if (!kk) return false;
    if (g === kk) return true;
    const a = Number(g), b = Number(kk);
    return isFinite(a) && isFinite(b) && Math.abs(a - b) < 1e-9;
  });
}
function dzFindResult_(code, work, row) {
  const sh = sheet_(SpreadsheetApp.getActive(), 'Результаты');
  if (!sh || sh.getLastRow() < 2) return null;
  const v = sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues();
  for (let i = 0; i < v.length; i++)
    if (String(v[i][0]).trim() === code && String(v[i][1]).trim() === work && (!row || !v[i][2] || Number(v[i][2]) === row))
      return {right: Number(v[i][4]) || 0, total: Number(v[i][5]) || 0};
  return null;
}
function dzLoad_(code, work, row) {
  if (code !== 'demo') {
    const st = findStudentRow_(code);
    if (!st || st.row[11] === false) return {error: 'not_found'};
  }
  const rows = dzRows_(work);
  if (!rows.length) return {error: 'no_work'};
  const done = code === 'demo' ? null : dzFindResult_(code, work, row);
  return {ok: true, title: rows[0].title || 'Домашнее задание', tasks: rows.map(r => ({n: r.n, lv: r.lv, q: r.q})), done: done};
}
function dzSubmit_(code, b) {
  const work = String(b.work || '').trim(), rows = dzRows_(work);
  if (!rows.length) return {error: 'no_work'};
  const demo = code === 'demo', row = Number(b.row) || 0;
  const ans = b.answers && typeof b.answers === 'object' ? b.answers : {};
  if (!demo) { const was = dzFindResult_(code, work, row); if (was) return {ok: true, right: was.right, total: was.total, again: true}; }
  const wrong = [], given = {};
  let right = 0;
  rows.forEach(r => {
    const g = String(ans[r.n] == null ? '' : ans[r.n]).slice(0, 100);
    given[r.n] = g;
    if (dzSame_(g, r.a)) right++; else wrong.push(r.n + (r.lv ? ' (ур. ' + r.lv + ')' : ''));
  });
  const total = rows.length;
  if (demo) return {ok: true, right: right, total: total, demo: true};
  const minutes = Math.max(0, Math.min(600, Math.round((Number(b.seconds) || 0) / 60)));
  const now = new Date();
  sheet_(SpreadsheetApp.getActive(), 'Результаты').appendRow([code, work, row || '', now, right, total, wrong.join(', '), minutes, JSON.stringify(given)]);
  const sh = SpreadsheetApp.getActive().getSheetByName('Домашка');
  if (sh && row >= 2 && row <= sh.getLastRow()) {
    if (sh.getMaxColumns() < 16) sh.insertColumnsAfter(sh.getMaxColumns(), 16 - sh.getMaxColumns());
    const v = sh.getRange(row, 1, 1, 16).getValues()[0];
    if (String(v[0]).trim() === code && !['checked', 'submitted'].includes(String(v[10] || ''))) {
      sh.getRange(1, 11, 1, 6).setValues([SHEETS['Домашка'].slice(10)]);
      sh.getRange(row, 11, 1, 6).setValues([['checked', String(v[11] || ''), 'Автопроверка: ' + right + ' из ' + total, String(v[13] || ''), now, Utilities.getUuid()]]);
      sh.getRange(row, 7, 1, 2).setValues([[true, now]]);
    }
  }
  dzNotify_(code, rows[0].title || work, right, total, wrong, minutes);
  return {ok: true, right: right, total: total};
}
function dzNotify_(code, title, right, total, wrong, minutes) {
  try {
    const st = findStudentRow_(code), who = st ? String(st.row[1] || '').trim() : code;
    const to = Session.getEffectiveUser().getEmail();
    if (!to) return;
    MailApp.sendEmail({
      to: to,
      subject: 'ДЗ сдано · ' + who + ' · ' + right + ' из ' + total,
      body: who + ' сдал(а) «' + title + '».\nВерно: ' + right + ' из ' + total + '.\nНеверные: ' + (wrong.length ? wrong.join(', ') : 'нет') +
            '\nВремя: ' + minutes + ' мин.\n\nКабинет ученика: ' + PAGE + '#lilia-' + code
    });
  } catch (e) { console.error('dzNotify_: ' + e); }
}

function telegramHash_(bytes) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, bytes)
    .map(b => ('0' + ((b + 256) % 256).toString(16)).slice(-2)).join('');
}
function telegramApi_(method, body, token) {
  token = token || PropertiesService.getScriptProperties().getProperty('TG_BOT_TOKEN');
  if (!token) return {ok: false, error: 'telegram_not_configured'};
  try {
    const res = UrlFetchApp.fetch('https://api.telegram.org/bot' + token + '/' + method, {
      method: 'post', contentType: 'application/json', payload: JSON.stringify(body || {}),
      muteHttpExceptions: true, followRedirects: false
    });
    const data = JSON.parse(res.getContentText());
    if (res.getResponseCode() === 200 && data.ok === true) return {ok: true, result: data.result};
    return {ok: false, error: data.error_code === 403 ? 'telegram_blocked' : 'telegram_api_error'};
  } catch (err) {
    // Ошибки UrlFetch могут содержать URL с токеном. Не возвращаем и не логируем их.
    return {ok: false, error: 'telegram_network_error'};
  }
}
function configureTelegram() {
  const ui = SpreadsheetApp.getUi();
  const a = ui.prompt('Подключить Telegram-бота', 'Вставь токен от @BotFather. Он сохранится только в настройках скрипта, не на сайте.', ui.ButtonSet.OK_CANCEL);
  if (a.getSelectedButton() !== ui.Button.OK) return;
  const token = a.getResponseText().trim();
  if (!/^\d+:[A-Za-z0-9_-]{20,}$/.test(token)) { ui.alert('Не похоже на токен Telegram-бота.'); return; }
  const me = telegramApi_('getMe', {}, token), hook = telegramApi_('getWebhookInfo', {}, token);
  if (!me.ok || !me.result.is_bot || !/^[a-zA-Z0-9_]{5,32}$/.test(me.result.username || '') || !hook.ok) {
    ui.alert('Не получилось проверить бота. Проверь токен и повтори.'); return;
  }
  if (hook.result.url) {
    ui.alert('Этот бот уже подключён к другому сервису через webhook. Создай отдельного бота для кабинета, чтобы не нарушить прежнее подключение.'); return;
  }
  const props = PropertiesService.getScriptProperties(), previous = props.getProperty('TG_BOT_TOKEN');
  props.setProperties({TG_BOT_TOKEN: token, TG_BOT_USERNAME: me.result.username, TG_SPREADSHEET_ID: SpreadsheetApp.getActive().getId()});
  ['Telegram', 'Telegram-ссылки', 'Telegram-файлы'].forEach(n => sheet_(telegramSS_(), n));
  if (previous && previous !== token) {
    telegramRows_('Telegram').forEach(r => telegramSS_().getSheetByName('Telegram').getRange(r.row, 2, 1, 5).setValues([['', false, false, '', '']]));
    const links = sheet_(telegramSS_(), 'Telegram-ссылки');
    if (links.getLastRow() >= 2) links.getRange(2, 1, links.getLastRow() - 1, 3).clearContent();
  }
  if (previous !== token) props.deleteProperty('TG_OFFSET');
  ScriptApp.getProjectTriggers().filter(t => ['pollTelegramUpdates', 'scanTelegramHomework'].includes(t.getHandlerFunction()))
    .forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('pollTelegramUpdates').timeBased().everyMinutes(1).create();
  ScriptApp.newTrigger('scanTelegramHomework').timeBased().everyMinutes(5).create();
  ui.alert('Бот @' + me.result.username + ' подключён. Теперь сделай новую версию развёртывания Apps Script. Ученики подключают уведомления кнопкой в своём кабинете.');
}
function telegramConnect_(code) {
  if (!telegramAvailable_()) return {error: 'telegram_not_configured'};
  const sh = sheet_(telegramSS_(), 'Telegram-ссылки'), now = Date.now();
  telegramRows_('Telegram-ссылки').filter(r => String(r.v[1]) === code || !(new Date(r.v[2]).getTime() > now))
    .sort((a, b) => b.row - a.row).forEach(r => sh.deleteRow(r.row));
  const token = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '').slice(0, 16);
  sh.appendRow([telegramHash_(token), code, new Date(now + 10 * 60 * 1000)]);
  return {ok: true, url: 'https://t.me/' + PropertiesService.getScriptProperties().getProperty('TG_BOT_USERNAME') + '?start=' + token};
}
function telegramSeedAgain_(code) {
  const sh = sheet_(telegramSS_(), 'Telegram-файлы');
  telegramRows_('Telegram-файлы').filter(r => String(r.v[1]) === code)
    .sort((a, b) => b.row - a.row).forEach(r => sh.deleteRow(r.row));
}
function telegramAdmin_(code, op, b) {
  const row = telegramBinding_(code);
  if (!row) return {error: 'telegram_not_connected'};
  const sh = sheet_(telegramSS_(), 'Telegram');
  if (op === 'telegramReset') {
    sh.getRange(row.row, 2, 1, 5).setValues([['', false, false, '', '']]);
    const links = sheet_(telegramSS_(), 'Telegram-ссылки');
    telegramRows_('Telegram-ссылки').filter(r => String(r.v[1]) === code).sort((a, b) => b.row - a.row).forEach(r => links.deleteRow(r.row));
    telegramSeedAgain_(code);
  } else {
    sh.getRange(row.row, 4).setValue(b.paused === true);
    // После тихого редактирования новое содержимое станет исходным эталоном.
    if (b.paused !== true) telegramSeedAgain_(code);
  }
  return {ok: true};

function telegramSend_(code, text, fileUrl) {

function telegramHandleUpdate_(update) {
  const m = update && update.message;
  if (!m || !m.chat || m.chat.type !== 'private' || !m.from || m.from.is_bot || !/^\d+$/.test(String(m.chat.id))) return;
  const text = String(m.text || '').trim(), chat = String(m.chat.id);
  if (/^\/stop(?:@\w+)?(?:\s|$)/i.test(text)) {
    if (chat === telegramTeacherChat_()) {
      PropertiesService.getScriptProperties().deleteProperty('TG_TEACHER_CHAT');
      telegramApi_('sendMessage', {chat_id: chat, text: 'Уведомления преподавателю отключены. Подключить заново: меню таблицы → Кабинет → Подключить Telegram преподавателя.'});
      return;

    telegramRows_('Telegram').filter(r => String(r.v[1]) === chat).forEach(r => sh.getRange(r.row, 2, 1, 5).setValues([['', false, false, '', '']]));

    return;
  }

    if (/^\/start(?:@\w+)?$/i.test(text)) telegramApi_('sendMessage', {chat_id: chat, text: 'Открой свой кабинет на lilimath.ru и нажми «Подключить Telegram». Затем вернись сюда по полученной ссылке и нажми «Старт».'});

  const link = telegramRows_('Telegram-ссылки').find(r => String(r.v[0]) === hash && new Date(r.v[2]).getTime() > Date.now());

function pollTelegramUpdates() {
  if (!telegramAvailable_()) return;
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) return;
  try {
    const props = PropertiesService.getScriptProperties();
    const res = telegramApi_('getUpdates', {offset: Number(props.getProperty('TG_OFFSET') || 0), timeout: 0, limit: 30, allowed_updates: ['message']});
    if (!res.ok || !Array.isArray(res.result)) return;
    for (const update of res.result) {
      if (!Number.isSafeInteger(update.update_id)) continue;
      telegramHandleUpdate_(update);
      props.setProperty('TG_OFFSET', String(update.update_id + 1));
    }
  } finally { lock.releaseLock(); }
}
function telegramSiteUrl_(value) {
  let url = String(value || '').trim();
  if (!url) return '';
  if (!/^[a-z]+:/i.test(url) && !url.startsWith('//')) url = 'https://lilimath.ru/' + url.replace(/^\/+/, '');
  const m = /^https:\/\/lilimath\.ru\/([^?#]*)(?:[?#].*)?$/i.exec(url);
  if (!m || /(?:^|\/)\.{1,2}(?:\/|$)|\\|%2e|%2f|%5c/i.test(m[1]) || !/\.(?:html?|pdf)$/i.test(m[1])) return '';
  // Личная ссылка на тест из конструктора: параметры нужны, иначе quiz.html откроется пустым.
  const q = /^https:\/\/lilimath\.ru\/quiz\.html(\?q=[\w%.-]+&v=\d+&a=[a-zA-Z0-9-]+)$/.exec(url);
  if (q) return 'https://lilimath.ru/quiz.html' + q[1];
  return 'https://lilimath.ru/' + m[1];
}
function telegramWatchList_() {
  const linked = {};
  telegramRows_('Telegram').forEach(r => { const code = String(r.v[0]); if (r.v[2] === true && telegramStudent_(code)) linked[code] = true; });
  const list = {}, hw = {}, add = (code, value, title) => {

    list[key] = {key: key, code: code, url: url, title: String(title || 'Домашнее задание').slice(0, 140)};
  };
  telegramRows_('Домашка').forEach(r => {

function telegramFetchFile_(url) {
  try {
    const r = UrlFetchApp.fetch(url, {muteHttpExceptions: true, followRedirects: false});
    const status = r.getResponseCode();
    if (status !== 200) return {error: 'http_' + status};
    const bytes = r.getContent();
    if (!bytes.length || bytes.length > 8 * 1024 * 1024) return {error: 'file_size'};
    return {hash: telegramHash_(bytes)};
  } catch (err) { return {error: 'fetch_failed'}; }
}
function scanTelegramHomework() {
  if (!telegramAvailable_()) return;
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) return;
  try {
    const props = PropertiesService.getScriptProperties(), sh = sheet_(telegramSS_(), 'Telegram-файлы');
    const list = telegramWatchList_(), wanted = {};
    list.forEach(x => { wanted[x.key] = true; });
    // Снятые или выполненные назначения не должны оставлять старый эталон навсегда.
    telegramRows_('Telegram-файлы').filter(r => !wanted[String(r.v[0])]).sort((a, b) => b.row - a.row).forEach(r => sh.deleteRow(r.row));
    const saved = {};
    telegramRows_('Telegram-файлы').forEach(r => { saved[String(r.v[0])] = r; });
    const urls = Array.from(new Set(list.map(x => x.url))).sort();
    if (!urls.length) return;
    // Не более 40 разных файлов за проход; курсор не даёт большому списку застрять на первых.
    const start = Number(props.getProperty('TG_SCAN_CURSOR') || 0) % urls.length;
    const selected = new Set(Array.from({length: Math.min(40, urls.length)}, (_, i) => urls[(start + i) % urls.length]));
    const fetched = {}, changes = {};
    const put = (item, hash, error) => {
      const values = [item.key, item.code, item.url, hash, new Date(), error || ''];
      if (saved[item.key]) sh.getRange(saved[item.key].row, 1, 1, 6).setValues([values]);
      else { sh.appendRow(values); saved[item.key] = {row: sh.getLastRow(), v: values}; }
    };
    list.filter(x => selected.has(x.url)).forEach(item => {
      const file = fetched[item.url] || (fetched[item.url] = telegramFetchFile_(item.url));
      const prev = saved[item.key], old = prev ? String(prev.v[3] || '') : '';
      if (file.error) { put(item, old, file.error); return; }
      const binding = telegramBinding_(item.code);
      if (!old || old === file.hash || (binding && binding.v[3] === true)) { put(item, file.hash, ''); return; }
      (changes[item.code] = changes[item.code] || []).push({item: item, hash: file.hash});
    });
    Object.keys(changes).forEach(code => {
      const batch = changes[code], titles = Array.from(new Set(batch.map(x => x.item.title)));
      const sent = telegramSend_(code, 'Обновилось домашнее задание\n\n' + titles.join('\n').slice(0, 2400), batch.length === 1 ? batch[0].item.url : '');
      batch.forEach(x => put(x.item, sent.ok ? x.hash : String(saved[x.item.key].v[3] || ''), sent.error));
    });
    props.setProperty('TG_SCAN_CURSOR', String((start + selected.size) % urls.length));
  } finally { lock.releaseLock(); }
}


/* Homework workflow v7.6. All writes run under the existing script lock. */
function workflowHw_(code, b, teacher) {
  const sh = SpreadsheetApp.getActive().getSheetByName('Домашка');
  const row = Number(b.id);
  if (!sh || !Number.isInteger(row) || row < 2 || row > sh.getLastRow()) return {error: 'changed'};
  if (sh.getMaxColumns() < 16) sh.insertColumnsAfter(sh.getMaxColumns(), 16 - sh.getMaxColumns());
  const v = sh.getRange(row, 1, 1, 16).getValues()[0];
  if (String(v[0]).trim() !== code || String(v[1]).trim() !== String(b.text || '').trim()) return {error: 'changed'};
  if (String(v[15] || '') !== String(b.revision || '')) return {error: 'changed'};
  const allowed = teacher ? ['checked', 'revise'] : ['solving', 'submitted', 'help'];
  const status = String(b.status || '');
  if (!allowed.includes(status)) return {error: 'bad_value'};
  const previous = String(v[10] || '');
  if (!teacher && ['checked', 'submitted'].includes(previous)) return {error: 'changed'};
  let note = String(b.note === undefined ? v[11] || '' : b.note).trim().slice(0, 2000);
  const feedback = teacher ? String(b.feedback || '').trim().slice(0, 2000) : String(v[12] || '');
  if (teacher && status === 'revise' && !feedback) return {error: 'bad_value'};
  let solutionUrl = String(v[13] || '');
  if (!teacher && b.photo) {
    const p = b.photo;
    if (!['image/jpeg', 'image/png', 'image/webp', 'application/pdf'].includes(String(p.mime)) || typeof p.data !== 'string' || p.data.length > 6990520 || !/^[A-Za-z0-9+/]*={0,2}$/.test(p.data)) return {error: 'bad_value'};
    const bytes = Utilities.base64Decode(p.data);
    if (!bytes.length || bytes.length > 5 * 1024 * 1024) return {error: 'bad_value'};
    const folders = DriveApp.getFoldersByName('Кабинет 25-25 · решения учеников');
    const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder('Кабинет 25-25 · решения учеников');
    const name = String(p.name || 'решение').replace(/[\\/:*?"<>|]/g, '_').slice(0, 100);
    const file = folder.createFile(Utilities.newBlob(bytes, p.mime, code + ' — ' + name));
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    solutionUrl = file.getUrl();
  }
  if (!teacher && status === 'submitted' && !note && !solutionUrl) return {error: 'bad_value'};
  const now = new Date(), revision = Utilities.getUuid();
  sh.getRange(1, 11, 1, 6).setValues([SHEETS['Домашка'].slice(10)]);
  sh.getRange(row, 11, 1, 6).setValues([[status, note, feedback, solutionUrl, now, revision]]);
  const done = status === 'submitted' || status === 'checked';
  sh.getRange(row, 7, 1, 2).setValues([[done, done ? now : '']]);
  return {ok: true, hw: {status: status, studentNote: note, feedback: feedback, solutionUrl: solutionUrl, revision: revision, done: done}};
}

/* ================= Напоминания о занятиях в Telegram =================

 * шлёт за сутки и за час через уже работающую telegramSend_. Отметки об отправке — лист «Напоминания». */
const LR_LESSON_MS = 60 * 60000;
const LR_BREAK_MS = 60 * 60000;
const LR_MAX_IN_A_ROW = 3;
const LR_TRANSFER_DAYS = 35;
const LR_MAX_DAILY_MS = 7 * 60 * 60000;
const LR_BREAK_LOOKAROUND_MS = LR_MAX_IN_A_ROW * (LR_LESSON_MS + LR_BREAK_MS);
function lrSamaraDay_(time) {
  const offset = LR_SAMARA_OFF * 60000;
  return Math.floor((time + offset) / 86400000) * 86400000 - offset;
}
function lrAvailability_() {
  const raw = PropertiesService.getScriptProperties().getProperty('LESSON_AVAILABILITY');
  return raw ? JSON.parse(raw) : {week: Array(7).fill(null), blocks: []};
}
function lrValidateAvailability_(value) {
  const valid = (start, end) => Number.isInteger(start) && Number.isInteger(end) &&
    start >= 0 && end <= 1440 && end > start;
  if (!value || !Array.isArray(value.week) || value.week.length !== 7 ||
      value.week.some(row => row !== null && (!Array.isArray(row) || !valid(row[0], row[1]))) ||
      !Array.isArray(value.blocks) || value.blocks.length > 100) throw new Error('bad_value');
  const interval = row => {
    if (!Array.isArray(row) || !/^\d{4}-\d{2}-\d{2}$/.test(row[0]) || !valid(row[1], row[2])) throw new Error('bad_value');
    const day = Date.parse(row[0] + 'T00:00:00Z');
    if (!Number.isFinite(day) || new Date(day).toISOString().slice(0, 10) !== row[0]) throw new Error('bad_value');
    return row.slice(0, 3);
  };
  const blocks = value.blocks.map(interval);
  const extra = value.extra || [];
  if (!Array.isArray(extra) || extra.length > 100) throw new Error('bad_value');
  const byDate = new Map();
  for (const row of extra.map(interval)) byDate.set(row[0], row);
  return {week: value.week.map(row => row && row.slice(0, 2)), blocks, extra: Array.from(byDate.values())};
}
function lrWorkHours_(local, availability) {
  const date = local.toISOString().slice(0, 10);
  const regular = availability.week[local.getUTCDay()];
  const extra = (availability.extra || []).find(row => row[0] === date);
  if (!extra) return regular;
  return regular ? [regular[0], Math.max(regular[1], extra[2])] : [extra[1], extra[2]];
}
function lrFits_(time, availability) {
  const day = lrSamaraDay_(time), local = new Date(day + LR_SAMARA_OFF * 60000);
  const work = lrWorkHours_(local, availability), date = local.toISOString().slice(0, 10);
  if (!work) return false;
  const start = (time - day) / 60000, end = start + LR_LESSON_MS / 60000;
  return start >= work[0] && end <= work[1] &&
    !availability.blocks.some(row => row[0] === date && row[1] < end && row[2] > start);
}
function lrBusy_(ss, after, before, skipCode, skipFrom, weekly = false) {
  const sh = ss.getSheetByName('Ученики');
  if (!sh || sh.getLastRow() < 2) return [];
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 12).getValues();
  const times = sh.getRange(2, 6, rows.length, 1).getDisplayValues();
  const days = Math.ceil((before - lrSamaraDay_(after)) / 86400000);
  const busy = [];
  rows.forEach((r, i) => {
    const code = String(r[0] || '').trim();
    if (!code || r[11] === false) return;
    const slots = lrParse_(r[4], times[i][0], 'Europe/Samara').slots;
    const moves = lrConfig_(code).moves;
    const skipped = new Set(moves.map(m => m.from));
    const replaces = start => code === skipCode && (weekly ? (start - skipFrom) % (7 * 86400000) === 0 : start === skipFrom);
    const regular = lrUpcoming_(slots, after, days).filter(t => !skipped.has(t) && !replaces(t));
    const moved = moves.filter(m => !replaces(m.from)).map(m => m.to);
    regular.concat(moved).forEach(t => { if (t > after && t < before) busy.push(t); });
  });
  return busy.sort((a, b) => a - b);
}
function lrMoveError_(time, busy) {
  const next = busy.findIndex(start => start >= time);
  const index = next === -1 ? busy.length : next;
  if (time - busy[index - 1] < LR_LESSON_MS || busy[index] - time < LR_LESSON_MS) return 'lesson_conflict';
  let count = 1, previous = time;
  for (let i = index - 1; i >= 0 && previous - busy[i] < LR_LESSON_MS + LR_BREAK_MS; i--) {
    if (++count > LR_MAX_IN_A_ROW) return 'lesson_break';
    previous = busy[i];
  }
  previous = time;
  for (let i = index; i < busy.length && busy[i] - previous < LR_LESSON_MS + LR_BREAK_MS; i++) {
    if (++count > LR_MAX_IN_A_ROW) return 'lesson_break';
    previous = busy[i];
  }
  const day = lrSamaraDay_(time), end = day + 86400000;
  let daily = LR_LESSON_MS;
  for (const start of busy) {
    if (start >= end) break;
    daily += Math.max(0, Math.min(end, start + LR_LESSON_MS) - Math.max(day, start));
    if (daily > LR_MAX_DAILY_MS) return 'lesson_day_limit';
  }
  return '';
}
function lrWindows_(availability, busy, now, end, expected) {
  const windows = [];
  for (let day = lrSamaraDay_(now); day < end; day += 86400000) {
    const work = lrWorkHours_(new Date(day + LR_SAMARA_OFF * 60000), availability);
    if (!work) continue;
    for (let t = day + Math.ceil(work[0] / 60) * LR_LESSON_MS; t + LR_LESSON_MS <= day + work[1] * 60000; t += LR_LESSON_MS) {
      if (t > now && t !== expected && lrFits_(t, availability) &&
          !lrMoveError_(t, busy)) windows.push(t);
    }
  }
  return windows;
}
function lrPublicRequest_(request) {
  if (!request) return null;
  const {id, status, kind, from, expected, to} = request;
  return {id, status, kind, from, expected, to};
}
function lrPupilOptions_(ss, code, st, kind, now) {
  const cfg = lrConfig_(code), day = lrSamaraDay_(now), end = day + LR_TRANSFER_DAYS * 86400000;
  const parsed = lrParse_(st.row[4], st.time, 'Europe/Samara');
  const regular = lrUpcoming_(parsed.slots, now, 8);
  const expected = Math.min(...lrApplyMoves_(regular, cfg.moves, now));
  if (!Number.isFinite(expected)) return {options: [], request: lrPublicRequest_(cfg.request)};
  const from = cfg.moves.find(move => move.to === expected)?.from ?? expected;
  return {from, expected, options: lrRequestWindows_(ss, code, st, cfg, kind, from, expected, now, end), request: lrPublicRequest_(cfg.request)};
}
function lrRequestWindows_(ss, code, st, cfg, kind, from, expected, now, end) {
  const weekly = kind === 'weekly', week = 7 * 86400000;
  const day = lrSamaraDay_(now);
  const slots = lrParse_(st.row[4], st.time, 'Europe/Samara').slots;
  const move = cfg.moves.find(item => item.from === from);
  if (expected <= now || (move?.to ?? from) !== expected ||
      (!move && !lrUpcoming_(slots, now, 8).includes(from))) return [];
  if (weekly && cfg.moves.some(item => Math.max(item.from, item.to) > now && (item.from - from) % week === 0)) return [];
  const busy = lrBusy_(ss, day - LR_BREAK_LOOKAROUND_MS, end + LR_BREAK_LOOKAROUND_MS, code, from, weekly);
  const availability = lrAvailability_();
  if (!weekly) return lrWindows_(availability, busy, now, end, expected);
  const base = {...availability, extra: []};
  const firstWeek = lrWindows_(base, busy, now, day + week, expected);
  return firstWeek.filter(start => {
    if ((start - from) % week === 0) return false;
    for (let t = start + week; t < end; t += week) {
      if (!lrFits_(t, base) || lrMoveError_(t, busy)) return false;
    }
    return true;
  });
}
function lrRequest_(ss, code, st, body, now) {
  const cfg = lrConfig_(code);
  if (cfg.request?.status === 'pending') return {error: 'lesson_request_pending'};
  const kind = body.kind === 'weekly' ? 'weekly' : 'once';
  const {from, expected, options} = lrPupilOptions_(ss, code, st, kind, now);
  if (body.from !== from || body.expected !== expected || !options.includes(body.to)) return {error: 'lesson_request_stale'};
  cfg.request = {id: Utilities.getUuid(), status: 'pending', kind, from, expected, to: body.to};
  lrSave_(code, cfg);
  return {ok: true, request: lrPublicRequest_(cfg.request)};
}
function lrDecideRequest_(ss, code, st, body, now) {
  const cfg = lrConfig_(code), request = cfg.request;
  if (!request || request.id !== body.id) return {error: 'changed'};
  if (request.status !== 'pending') return {ok: true};
  const applied = request.kind === 'weekly' && request.appliedDays && st.row[4] === request.appliedDays;
  if (applied) {
    request.status = 'approved';
    lrSave_(code, cfg);
    return {ok: true};
  }
  if (body.approve === true) {
    if (st.row[11] === false) return {error: 'inactive'};
    const {kind, from, expected, to} = request;
    const end = lrSamaraDay_(now) + LR_TRANSFER_DAYS * 86400000;
    if (!lrRequestWindows_(ss, code, st, cfg, kind, from, expected, now, end).includes(to)) return {error: 'lesson_request_stale'};
    if (kind === 'weekly') {
      const offset = LR_SAMARA_OFF * 60000;
      const old = new Date(from + offset), next = new Date(to + offset);
      const days = String(st.row[4] || '');
      const city = days.match(/^\s*\[[^\]]+\]\s*/)?.[0] || '';
      const zone = days.includes('|') ? days.slice(days.indexOf('|')) : '';
      const names = Object.keys(LR_DAYS);
      const slots = lrParse_(days, st.time, 'Europe/Samara').slots.map(slot =>
        slot.wd === old.getUTCDay() && slot.h === old.getUTCHours() && slot.m === old.getUTCMinutes()
          ? {wd: next.getUTCDay(), h: next.getUTCHours(), m: next.getUTCMinutes()} : slot);
      const schedule = slots.map(slot => names[slot.wd] + ' ' + String(slot.h).padStart(2, '0') + ':' + String(slot.m).padStart(2, '0')).join(', ');
      request.appliedDays = city + schedule + (zone ? ' ' + zone : '');
      lrSave_(code, cfg);
      ss.getSheetByName('Ученики').getRange(st.index, STUDENT_COLS.days).setValue(request.appliedDays);
    } else if (kind === 'once') {
      cfg.moves = cfg.moves.filter(move => move.from !== from);
      if (to !== from) cfg.moves.push({from, to});
    }
  }
  request.status = body.approve === true ? 'approved' : 'rejected';
  lrSave_(code, cfg);
  return {ok: true};
}
function lrMinutes_(value) {
  if (!Array.isArray(value) || value.length > 3 ||
      value.some(n => !Number.isInteger(n) || n < 15 || n > 2880)) throw new Error('bad_value');
  return Array.from(new Set(value)).sort((a, b) => b - a);
}
function lrDefaults_() {
  const raw = PropertiesService.getScriptProperties().getProperty('LESSON_DEFAULTS');
  return raw ? JSON.parse(raw) : [1440, 60];
}
function lrConfig_(code) {
  const raw = PropertiesService.getScriptProperties().getProperty('LESSON_' + code);
  return raw ? JSON.parse(raw) : {minutes: null, moves: []};
}
function lrSave_(code, config) {
  const cutoff = Date.now() - 3600000;
  config.moves = config.moves.filter(m => Math.max(m.from, m.to) > cutoff);
  const props = PropertiesService.getScriptProperties(), key = 'LESSON_' + code;
  if (config.minutes === null && !config.moves.length && !config.request) props.deleteProperty(key);
  else props.setProperty(key, JSON.stringify(config));
}
function lrApplyMoves_(regular, moves, now) {
  const skipped = new Set(moves.map(m => m.from));
  return Array.from(new Set(regular.filter(t => !skipped.has(t)).concat(moves.map(m => m.to).filter(t => t > now))));
}
const LR_WINDOW_MIN = 15;   // триггер каждые 10 минут, окно 15: хотя бы одно срабатывание гарантировано
const LR_SAMARA_OFF = 4 * 60;
const LR_DAYS = {'вс': 0, 'пн': 1, 'вт': 2, 'ср': 3, 'чт': 4, 'пт': 5, 'сб': 6};
const LR_DAY_NAMES = ['воскресенье', 'понедельник', 'вторник', 'среда', 'четверг', 'пятница', 'суббота'];
const LR_MONTHS = ['января', 'февраля', 'марта', 'апреля', 'мая', 'июня', 'июля', 'августа', 'сентября', 'октября', 'ноября', 'декабря'];

function lrParse_(days, time, tzDefault) {
  let raw = String(days || '').replace(/^\s*\[[^\]]{1,40}\]\s*/, '');
  const k = raw.indexOf('|');
  const a = k < 0 ? raw : raw.slice(0, k);
  let tz = k < 0 ? '' : raw.slice(k + 1).trim();
  if (tz) { try { Utilities.formatDate(new Date(), tz, 'HH'); } catch (e) { tz = ''; } }
  const slots = [], re = /(пн|вт|ср|чт|пт|сб|вс)\s*(\d{1,2})[:.](\d{2})/gi;
  let m;
  while ((m = re.exec(a))) slots.push({wd: LR_DAYS[m[1].toLowerCase()], h: +m[2], m: +m[3]});
  if (!slots.length) {   // старый формат: «ср, сб» и время в отдельной колонке
    const ds = a.toLowerCase().match(/пн|вт|ср|чт|пт|сб|вс/g) || [];
    const t = time instanceof Date ? Utilities.formatDate(time, tzDefault, 'HH:mm') : String(time || '');
    const tm = t.match(/(\d{1,2})[:.](\d{2})/);
    if (tm) ds.forEach(d => slots.push({wd: LR_DAYS[d], h: +tm[1], m: +tm[2]}));
  }
  return {slots: slots, tz: tz || 'Europe/Samara'};
}

/* Ближайшие занятия (мс UTC) в ближайшие 3 суток по самарскому времени. */
function lrUpcoming_(slots, nowMs, days = 3) {
  const out = [], sam = new Date(nowMs + LR_SAMARA_OFF * 60000);
  slots.forEach(s => {
    for (let off = 0; off <= days; off++) {
      const d = new Date(Date.UTC(sam.getUTCFullYear(), sam.getUTCMonth(), sam.getUTCDate() + off));
      if (d.getUTCDay() !== s.wd) continue;
      const t = Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate(), s.h, s.m) - LR_SAMARA_OFF * 60000;
      if (t > nowMs) out.push(t);
    }
  });
  return out;
}

function lrWhen_(ms, tz) {
  const d = new Date(ms);
  const dow = Number(Utilities.formatDate(d, tz, 'u')) % 7;   // 'u': 1=пн ... 7=вс
  return LR_DAY_NAMES[dow] + ', ' + Number(Utilities.formatDate(d, tz, 'd')) + ' ' +
    LR_MONTHS[Number(Utilities.formatDate(d, tz, 'M')) - 1] + ', ' + Utilities.formatDate(d, tz, 'HH:mm');
}

function sendLessonReminders() {
  if (!telegramAvailable_()) return;
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) return;
  try {
    const ss = telegramSS_(), ssTz = ss.getSpreadsheetTimeZone();
    const st = ss.getSheetByName('Ученики');
    if (!st || st.getLastRow() < 2) return;
    let log = ss.getSheetByName('Напоминания');
    if (!log) { log = ss.insertSheet('Напоминания'); log.getRange(1, 1, 1, 2).setValues([['Ключ', 'Отправлено']]); }
    const sentKeys = {};
    if (log.getLastRow() > 1) log.getRange(2, 1, log.getLastRow() - 1, 1).getValues().forEach(r => { sentKeys[String(r[0])] = true; });
    const connected = {};
    telegramRows_('Telegram').forEach(r => { if (r.v[2] === true && r.v[3] !== true) connected[String(r.v[0])] = true; });
    const now = Date.now(), defaults = lrDefaults_();
    st.getRange(2, 1, st.getLastRow() - 1, 12).getValues().forEach(r => {
      const code = String(r[0] || '').trim();
      if (!code || r[11] === false || !connected[code]) return;
      const P = lrParse_(r[4], r[5], ssTz);
      const cfg = lrConfig_(code), minutes = cfg.minutes === null ? defaults : cfg.minutes;
      lrApplyMoves_(lrUpcoming_(P.slots, now), cfg.moves, now).forEach(t => {
        const left = (t - now) / 60000;
        minutes.forEach(min => {
          if (!(left <= min && left > min - LR_WINDOW_MIN)) return;
          const key = code + '|' + t + '|' + (min === 1440 ? 'd1' : min === 60 ? 'h1' : min);
          if (sentKeys[key]) return;
          const text = 'Напоминание о занятии\n\n' + lrWhen_(t, P.tz) +
            (P.tz !== 'Europe/Samara' ? ' (по твоему времени)' : '');
          const res = telegramSend_(code, text);
          if (res.ok) { log.appendRow([key, new Date()]); sentKeys[key] = true; }
        });
      });
    });
    if (log.getLastRow() > 600) log.deleteRows(2, 300);   // старые отметки не нужны, занятия уже прошли
  } finally { lock.releaseLock(); }
}

function setupLessonReminders() {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'sendLessonReminders').forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('sendLessonReminders').timeBased().everyMinutes(10).create();
}


/* ================= Занятия по карте ================= */
function routeId_(id) {
  id = String(id || '');
  if (!/^(?:8A-(?:F\d{1,2}|\d{2,3})|(?:5|6|7A|7G|8G|9A|9G)-\d{2,3}|OGE-\d{2}-\d)$/.test(id)) throw new Error('bad_value');
  return id;
}
function routeMaterials_(id) {
  const raw = PropertiesService.getScriptProperties().getProperty('ROUTE_' + routeId_(id));
  return raw ? JSON.parse(raw) : {theory: '', homework: ''};
}
function routeSave_(b) {
  const links = {};
  for (const key of ['theory', 'homework']) {
    const url = String(b[key] || '').trim();
    if (url && (url.length > 1500 || !/^https:\/\/[^\s<>"']+$/i.test(url))) return {error: 'bad_value'};
    links[key] = url;
  }
  PropertiesService.getScriptProperties().setProperty('ROUTE_' + routeId_(b.id), JSON.stringify(links));
  return {ok: true, materials: links};
}
function routeUpload_(b) {
  const data = String(b.data || '');
  if (!data.startsWith('JVBERi0') || data.length > 11 * 1024 * 1024) return {error: 'bad_value'};
  const name = String(b.name || 'Материал.pdf').replace(/[\\/:*?"<>|]/g, '_').slice(0, 150);
  const it = DriveApp.getFoldersByName('lili.math · материалы занятий');
  const folder = it.hasNext() ? it.next() : DriveApp.createFolder('lili.math · материалы занятий');
  const file = folder.createFile(Utilities.newBlob(Utilities.base64Decode(data), 'application/pdf', name));
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return {ok: true, url: file.getUrl()};
}
function routeAssign_(ss, st, code, b) {
  if (st.row[11] === false) return {error: 'not_found'};
  const id = routeId_(b.id);
  const title = String(b.title || '').trim();
  if (!title || title.length > 200 || /^[=+@-]/.test(title)) return {error: 'bad_value'};
  const date = Utilities.formatDate(new Date(), 'Europe/Samara', 'yyyy-MM-dd');
  const key = date + '|' + id;
  const journal = sheet_(ss, 'Занятия по карте');
  let row = findRow_('Занятия по карте', v => String(v[0]).trim() === code && v[1] === key);
  if (!row) {
    const materials = routeMaterials_(id);
    journal.appendRow([code, key, date, title, materials.theory, materials.homework, false]);
    row = journal.getLastRow();
  }
  const entry = journal.getRange(row, 1, 1, 7).getValues()[0];
  if (entry[6] === true) return {ok: true, duplicate: true};
  if (entry[5] && !findRow_('Домашка', v => String(v[0]).trim() === code && v[16] === key)) {
    const hw = sheet_(ss, 'Домашка');
    hw.appendRow([code, 'Домашняя работа: ' + entry[3], '', '', entry[5], 'Открыть домашнюю работу', false, '', '', '', '', '', '', '', '', '', key]);
  }
  journal.getRange(row, 7).setValue(true);
  return {ok: true, theory: !!entry[4], homework: !!entry[5]};
}
