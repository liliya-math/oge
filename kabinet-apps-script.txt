function normProg_(p) {
  p = String(p || '');
  if (p === 'school') return 'school5';
  return (p === 'oge9' || /^school[5-8]$/.test(p) || /^((alg|geo|mix)[789])$/.test(p)) ? p : 'oge9';
}
function progOf_(row, f) {
  if (f && f.prog !== undefined) return normProg_(f.prog);
  return normProg_(row[13]);
}
function clsOfProg_(p) {
  const m = /^[a-z]*(\d)$/.exec(String(p || ''));
  return m ? Number(m[1]) : 9;
}

  if (p === 'oge9') return 31;

  if (p === 'geo7') return 11;

  return 15;
}

function ptsOf_(p, n) {
  if (p === 'oge9') return n >= 20 ? 2 : 1;
  if (p === 'mix9') return n <= 33 ? 1 : (n - 33 >= 20 ? 2 : 1);
  return 1;
}
function sheet_(ss, name) {
  let sh = ss.getSheetByName(name);
  if (!sh && SHEETS[name]) {
    sh = ss.insertSheet(name, 99);
    sh.getRange(1, 1, 1, SHEETS[name].length).setValues([SHEETS[name]]);
    sh.setFrozenRows(1);
  }
  return sh;
}

function setup() {
  const ss = SpreadsheetApp.getActive();
  Object.keys(SHEETS).forEach((name, i) => {
    let sh = ss.getSheetByName(name);
    if (!sh) sh = ss.insertSheet(name, i);
    const head = SHEETS[name];
    if (sh.getMaxColumns() < head.length) sh.insertColumnsAfter(sh.getMaxColumns(), head.length - sh.getMaxColumns());
    sh.getRange(1, 1, 1, head.length).setValues([head]);
    sh.setFrozenRows(1);
  });
  ['Sheet1', 'Лист1'].forEach(n => {
    const st = ss.getSheetByName(n);
    if (st && st.getLastRow() === 0 && ss.getSheets().length > 1) ss.deleteSheet(st);
  });
  const map = ss.getSheetByName('Карта');
  const rule = SpreadsheetApp.newDataValidation().requireValueInList(STATUSES, true).setAllowInvalid(false).build();
  map.getRange(2, 3, 300, 58).setDataValidation(rule);

  const hw = ss.getSheetByName('Домашка');
  hw.getRange(2, 7, 1000, 1).insertCheckboxes();
  hw.getRange(2, 4, 1000, 1).setNumberFormat('dd.MM.yyyy');
  hw.getRange(2, 8, 1000, 1).setNumberFormat('dd.MM.yyyy HH:mm');
  const hwSecRule = SpreadsheetApp.newDataValidation().requireValueInList(HW_SECS, true).setAllowInvalid(true).build();
  hw.getRange(2, 10, 1000, 1).setDataValidation(hwSecRule);
  ss.getSheetByName('Пробники').getRange(2, 2, 1000, 1).setNumberFormat('dd.MM.yyyy');
  ss.getSheetByName('Оценки').getRange(2, 2, 1000, 1).setNumberFormat('dd.MM.yyyy');
  const letters = ss.getSheetByName('Письма');
  letters.getRange(2, 4, 1000, 1).insertCheckboxes();
  letters.getRange(2, 2, 1000, 1).setNumberFormat('dd.MM.yyyy HH:mm');
  const sk = ss.getSheetByName('Навыки');
  sk.getRange(2, 3, 3000, 1).setDataValidation(rule);
  sk.getRange(2, 4, 3000, 1).setNumberFormat('dd.MM.yyyy');
  const us = ss.getSheetByName('Ученики');
  us.getRange(2, 12, 300, 1).insertCheckboxes();
  us.getRange(2, 6, 300, 1).setNumberFormat('@');
  us.getRange(2, 1, 300, 1).setNumberFormat('@');
  us.getRange(2, 18, 300, 1).setNumberFormat('@');
  us.getRange(2, 16, 300, 1).setNumberFormat('dd.MM.yyyy');
  if (us.getLastRow() >= 2) {
    const codes = us.getRange(2, 1, us.getLastRow() - 1, 1).getValues();
    us.getRange(2, 13, codes.length, 1).setValues(codes.map(c => [c[0] ? PAGE + '#' + c[0] : '']));
  }
  DriveApp.getRootFolder(); // запросить доступ к Диску сразу
  styleSheets();
  syncMaterials();
}

/* ================= материалы сайта ================= */
function syncMaterials() {
  const ss = SpreadsheetApp.getActive();
  let sh = ss.getSheetByName('Материалы');
  if (!sh) {
    sh = ss.insertSheet('Материалы');
    sh.getRange(1, 1, 1, 2).setValues([SHEETS['Материалы']]);
    sh.setFrozenRows(1);
  }
  let rz = ss.getSheetByName('Разборы');
  if (!rz) {
    rz = ss.insertSheet('Разборы');
    rz.getRange(1, 1, 1, 4).setValues([SHEETS['Разборы']]);
    rz.setFrozenRows(1);
  }
  sh.getRange(2, 1, Math.max(sh.getMaxRows() - 1, 1), 2).clearContent();
  sh.getRange(2, 1, MATERIALS.length, 2).setValues(MATERIALS);
  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInRange(sh.getRange(2, 1, MATERIALS.length, 1), true)
    .setAllowInvalid(false).build();
  rz.getRange(2, 2, 1000, 1).setDataValidation(rule);
}

function styleSheets() {
  const ss = SpreadsheetApp.getActive();
  const widths = {
    'Ученики': [110, 120, 130, 90, 110, 110, 260, 200, 320, 110, 110, 80, 380, 130, 130, 120, 160, 160, 70],
    'Карта': [110, 120].concat(Array(58).fill(54)),
    'Комментарии': [110, 90, 420, 260, 200],
    'Разборы': [110, 300, 320, 160],
    'Материалы': [300, 320],
    'Домашка': [110, 340, 110, 110, 260, 160, 90, 150, 300, 150],
    'Файлы': [110, 110, 260, 340],
    'Пробники': [110, 110, 220, 90],
    'Оценки': [110, 110, 140, 90, 300],
    'Письма': [110, 150, 520, 100],
    'Навыки': [110, 90, 130, 120]
  };
  const tabs = {'Ученики': C.cobalt, 'Карта': C.sage, 'Комментарии': C.amber, 'Разборы': C.miss, 'Материалы': '#9db8ff', 'Домашка': C.apricot, 'Файлы': '#9db8ff', 'Пробники': '#9db8ff', 'Оценки': C.sage, 'Письма': C.amber, 'Навыки': C.sage};
  Object.keys(SHEETS).forEach(name => {
    const sh = ss.getSheetByName(name);
    if (!sh) return;
    const cols = SHEETS[name].length, rows = sh.getMaxRows();
    if (sh.getMaxColumns() > cols) sh.deleteColumns(cols + 1, sh.getMaxColumns() - cols);
    sh.getBandings().forEach(b => b.remove());
    const all = sh.getRange(1, 1, rows, cols);
    all.setBackground(null).setFontColor(C.fg).setFontFamily('Inter').setFontSize(11).setVerticalAlignment('middle');
    all.applyRowBanding(SpreadsheetApp.BandingTheme.GREY, true, false)
       .setHeaderRowColor(C.cobalt).setFirstRowColor(C.bg).setSecondRowColor(C.bg2);
    const head = sh.getRange(1, 1, 1, cols);
    head.setFontWeight('bold').setFontColor(C.fg).setWrap(true)
        .setBorder(null, null, true, null, null, null, C.apricot, SpreadsheetApp.BorderStyle.SOLID_THICK);
    sh.setRowHeight(1, 44);
    if (rows > 1) sh.setRowHeights(2, rows - 1, 30);
    sh.getRange(2, 1, Math.max(rows - 1, 1), 1).setFontColor(C.muted).setFontFamily('Roboto Mono').setFontSize(10);
    (widths[name] || []).forEach((w, i) => sh.setColumnWidth(i + 1, w));
    sh.setHiddenGridlines(true);
    sh.setTabColor(tabs[name]);
    sh.setFrozenRows(1);
    if (name === 'Ученики' || name === 'Карта') {
      sh.setFrozenColumns(2);
      sh.getRange(2, 2, Math.max(rows - 1, 1), 1).setFontWeight('bold');
    }
  });

  const map = ss.getSheetByName('Карта');
  if (map) {
    const rng = map.getRange(2, 3, Math.max(map.getMaxRows() - 1, 1), 58);
    rng.setHorizontalAlignment('center').setFontWeight('bold').setFontSize(10);
    map.getRange(1, 3, 1, 58).setHorizontalAlignment('center').setFontSize(14);
    const colors = [['решает', C.sage, C.bg], ['путается', C.amber, C.bg], ['не решает', C.miss, '#2a1510'], ['не проходили', C.none, C.muted]];
    map.setConditionalFormatRules(colors.map(([v, bg, fg]) => SpreadsheetApp.newConditionalFormatRule()
      .whenTextEqualTo(v).setBackground(bg).setFontColor(fg).setRanges([rng]).build()));
  }
  const hw = ss.getSheetByName('Домашка');
  if (hw) {
    const n = Math.max(hw.getMaxRows() - 1, 1);
    hw.getRange(2, 2, n, 1).setWrap(true);
    hw.getRange(2, 3, n, 2).setHorizontalAlignment('center');
    hw.getRange(2, 7, n, 1).setHorizontalAlignment('center');
    hw.setConditionalFormatRules([SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied('=$G2=TRUE').setFontColor(C.muted).setStrikethrough(true)
      .setRanges([hw.getRange(2, 2, n, 1)]).build()]);
  }
  const pr = ss.getSheetByName('Пробники');
  if (pr) pr.getRange(2, 4, Math.max(pr.getMaxRows() - 1, 1), 1).setHorizontalAlignment('center').setFontWeight('bold').setFontColor(C.amber);
}

/* ================= ученики ================= */
function addStudent() {
  const ui = SpreadsheetApp.getUi();
  const a = ui.prompt('Новый ученик', 'Имя (как в кабинете), например: Маша', ui.ButtonSet.OK_CANCEL);
  if (a.getSelectedButton() !== ui.Button.OK || !a.getResponseText().trim()) return;
  const name = a.getResponseText().trim();
  const b = ui.prompt('Новый ученик', 'Кому — дательный падеж, например: Маше', ui.ButtonSet.OK_CANCEL);
  if (b.getSelectedButton() !== ui.Button.OK) return;
  const code = createStudent_(name, b.getResponseText().trim());
  showLinkFor_(name, code);
}

function createStudent_(name, dat, prog, cls) {
  const ss = SpreadsheetApp.getActive();
  if (!ss.getSheetByName('Ученики')) setup();
  const code = newCode_();
  const p = normProg_(prog);
  const goal = p === 'oge9' ? 22 : Math.round(goalMax_(p) * 0.8);
  ss.getSheetByName('Ученики').appendRow([code, name, dat || name, goal, 'ср, сб', '18:00',
    '', 'https://learnboard.ru', '', 8, 0, true, PAGE + '#' + code, p, '', '', '', '', cls || clsOfProg_(p)]);
  ss.getSheetByName('Карта').appendRow([code, name].concat(Array(58).fill('не проходили')));
  return code;
}

function showLink() {
  const sh = SpreadsheetApp.getActiveSheet();
  const row = sh.getActiveRange().getRow();
  if (row < 2) { SpreadsheetApp.getUi().alert('Выдели строку ученика на листе «Ученики» или «Карта».'); return; }
  const code = String(sh.getRange(row, 1).getValue()).trim();
  const st = findStudentRow_(code);
  if (!st) { SpreadsheetApp.getUi().alert('В этой строке нет кода ученика.'); return; }
  showLinkFor_(st.row[1], code);
}

function showLinkFor_(name, code) {
  const url = PAGE + '#' + code;
  const sh = SpreadsheetApp.getActive().getSheetByName('Ученики');
  const st = findStudentRow_(code);
  if (st) sh.getRange(st.index, 13).setValue(url);
  SpreadsheetApp.getUi().alert('Кабинет: ' + name, 'Личная ссылка (она же в столбце M на листе «Ученики»):\n\n' + url, SpreadsheetApp.getUi().ButtonSet.OK);
}

function setPassword() {
  const ui = SpreadsheetApp.getUi();
  const a = ui.prompt('Пароль для режима учителя', 'Не меньше 6 символов. Его будешь вводить на странице kabinet.html#lilia', ui.ButtonSet.OK_CANCEL);
  if (a.getSelectedButton() !== ui.Button.OK) return;
  const p = a.getResponseText().trim();
  if (p.length < 6) { ui.alert('Слишком короткий пароль. Нужно хотя бы 6 символов.'); return; }
  PropertiesService.getScriptProperties().setProperty('ADMIN', p);
  ui.alert('Пароль сохранён. Режим учителя: ' + PAGE + '#lilia');
}

function doGet(e) {
  const code = String((e && e.parameter && e.parameter.t) || '').trim();
  return json_(code ? studentData_(code, false) : {error: 'no_code'});
}

function doPost(e) {
  let body = {};
  try { body = JSON.parse(e.postData.contents); } catch (err) { return json_({error: 'bad_request'}); }
  if (body.admin !== undefined) return json_(admin_(body));
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const st = findStudentRow_(body.t);
    if (!st || st.row[11] === false) return json_({error: 'not_found'});
    const code = String(body.t).trim();
    if (body.letter !== undefined) return json_(saveLetter_(code, body.letter));
    return json_(setHwDone_(code, body.id, body.text, body.done));
  } finally { lock.releaseLock(); }
}

function saveLetter_(code, text) {
  const t = String(text || '').trim().slice(0, 2000);
  if (t.length < 2) return {error: 'bad_value'};
  let sh = SpreadsheetApp.getActive().getSheetByName('Письма');
  if (!sh) { sh = SpreadsheetApp.getActive().insertSheet('Письма', 99); sh.getRange(1, 1, 1, 4).setValues([SHEETS['Письма']]); sh.setFrozenRows(1); }
  sh.appendRow([code, new Date(), t, false]);
  notifyLetter_(code, t);
  return {ok: true};
}

// Письмо на почту владельца скрипта. Сбой отправки не мешает сохранению письма.
function notifyLetter_(code, text) {
  try {
    const st = findStudentRow_(code);
    const who = st ? String(st.row[1] || '').trim() : '';
    const to = Session.getEffectiveUser().getEmail();
    if (!to) return;
    MailApp.sendEmail({
      to: to,
      subject: 'Кабинет 25/25: новое письмо' + (who ? ' · ' + who : ''),
      body: 'Новое письмо от ученика' + (who ? ' ' + who : '') + ' (код ' + code + '):\n\n' + text +
            '\n\nОткрыть кабинет в режиме учителя: ' + PAGE + '#lilia-' + code
    });
  } catch (e) { console.error('notifyLetter_: ' + e); }
}

function setHwDone_(code, id, text, done) {
  const sh = SpreadsheetApp.getActive().getSheetByName('Домашка');
  const row = Number(id);
  if (!(row >= 2 && row <= sh.getLastRow())) return {error: 'no_task'};
  const v = sh.getRange(row, 1, 1, 2).getValues()[0];
  if (String(v[0]).trim() !== code || String(v[1]).trim() !== String(text || '').trim()) return {error: 'changed'};
  sh.getRange(row, 7, 1, 2).setValues([[!!done, done ? new Date() : '']]);
  return {ok: true};
}

/* ================= режим учителя ================= */
function admin_(b) {
  const pass = PropertiesService.getScriptProperties().getProperty('ADMIN');
  if (!pass) return {error: 'no_password'};
  if (String(b.admin || '') !== pass) { Utilities.sleep(700); return {error: 'bad_password'}; }
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = SpreadsheetApp.getActive();
    const code = String(b.t || '').trim();
    const need = () => { const st = findStudentRow_(code); if (!st) throw new Error('not_found'); return st; };
    const fresh = () => ({ok: true, d: studentData_(code, true)});
    switch (b.op) {
      case 'ping': return {ok: true};
      case 'list': return {ok: true, students: listStudents_()};
      case 'get': { need(); return fresh(); }
      case 'addStudent': {
        const name = String(b.name || '').trim();
        if (!name) return {error: 'no_name'};
        return {ok: true, code: createStudent_(name, String(b.dat || '').trim(), b.prog, Number(b.cls) || undefined)};
      }
      case 'student': {
        const st = need(), sh = ss.getSheetByName('Ученики'), f = b.fields || {};
        const pr = progOf_(st.row, f);
        Object.keys(f).forEach(k => {
          if (!STUDENT_COLS[k]) return;
          let v = f[k];
          if (k === 'work') v = (Array.isArray(v) ? v : String(v).split('\n')).map(s => String(s).trim()).filter(Boolean).join('; ');
          else if (k === 'prog') v = normProg_(v);
          else if (k === 'goal' || k === 'lessonsPlan' || k === 'lessonsDone') v = Math.max(0, Math.min(k === 'goal' ? goalMax_(pr) : 99, Number(v) || 0));
          else if (k === 'active') v = !!v;
          else if (k === 'bdate') v = /^\d{4}-\d{2}-\d{2}$/.test(String(v || '')) ? new Date(String(v) + 'T00:00:00') : '';
          else if (k === 'cls') v = Math.max(0, Math.min(11, Number(v) || 0));
          else v = String(v).trim();
          sh.getRange(st.index, STUDENT_COLS[k]).setValue(v);
        });
        if (f.name) { const m = findRow_('Карта', r => String(r[0]).trim() === code); if (m) ss.getSheetByName('Карта').getRange(m, 2).setValue(String(f.name).trim()); }
        return fresh();
      }
      case 'map': {
        const st = need();
        const n = Number(b.n), s = STATUS_RU[b.s];
        if (!(n >= 1 && n <= nMax_(progOf_(st.row))) || !s) return {error: 'bad_value'};
        const sh = ss.getSheetByName('Карта');
        let row = findRow_('Карта', r => String(r[0]).trim() === code);
        if (!row) { sh.appendRow([code, st.row[1]].concat(Array(25).fill('не проходили'))); row = sh.getLastRow(); }
        sh.getRange(row, n + 2).setValue(s);
        return fresh();
      }
      case 'skill': {
        need();
        const id = String(b.n || '').trim().toUpperCase(), s = STATUS_RU[b.s];
        if (!SKILL_RE.test(id) || !s) return {error: 'bad_value'};
        const sh = sheet_(ss, 'Навыки');
        const row = findRow_('Навыки', r => String(r[0]).trim() === code && String(r[1]).trim().toUpperCase() === id);
        if (b.s === 'new') { if (row) sh.deleteRow(row); }   // «не проверяли» = строки в таблице нет
        else if (row) sh.getRange(row, 3, 1, 2).setValues([[s, new Date()]]);
        else sh.appendRow([code, id, s, new Date()]);
        return fresh();
      }
      case 'note': {
        const st = need();
        const n = Number(b.n);
        if (!(n >= 1 && n <= nMax_(progOf_(st.row)))) return {error: 'bad_value'};
        const sh = ss.getSheetByName('Комментарии');
        const row = findRow_('Комментарии', r => String(r[0]).trim() === code && Number(r[1]) === n);
        const t = String(b.text || '').trim(), u = String(b.url || '').trim(), l = String(b.label || '').trim();
        if (!t && !u) { if (row) sh.deleteRow(row); }
        else if (row) sh.getRange(row, 3, 1, 3).setValues([[t, u, l]]);
        else sh.appendRow([code, n, t, u, l]);
        return fresh();
      }
      case 'hwAdd': {
        need();
        const t = String(b.text || '').trim();
        if (!t) return {error: 'bad_value'};
        const sec = HW_SEC_KEY[String(b.sec || '').trim().toLowerCase()] || '';
        const hwSh = ss.getSheetByName('Домашка');
        hwSh.appendRow([code, t, String(b.n || ''), ruDate_(b.due), String(b.url || ''), String(b.label || ''), false, '', String(b.file || ''), sec]);
        const rowNo = hwSh.getLastRow();
        (Array.isArray(b.files) ? b.files : []).slice(0, 10).forEach(f => {
          const u = String(f && f.url || '').trim();
          if (u) sheet_(ss, 'Файлы').appendRow([code, rowNo, String(f.name || 'PDF для печати').trim().slice(0, 120), u]);
        });
        return fresh();
      }
      case 'fileAdd': {
        need();
        const hwRow = Number(b.hw);
        const u = String(b.url || '').trim();
        if (!(hwRow >= 2) || !u) return {error: 'bad_value'};
        sheet_(ss, 'Файлы').appendRow([code, hwRow, String(b.name || 'PDF для печати').trim().slice(0, 120), u]);
        return fresh();
      }
      case 'fileDel': {
        need();
        const sh = sheet_(ss, 'Файлы'), row = Number(b.id);
        if (!(row >= 2 && row <= sh.getLastRow()) || String(sh.getRange(row, 1).getValue()).trim() !== code) return {error: 'changed'};
        sh.deleteRow(row);
        return fresh();
      }
      case 'upload': {
        const st = need();
        const name = String(b.fname || 'домашка.pdf').replace(/[\\/:*?"<>|]/g, '_');
        if (!b.data) return {error: 'bad_value'};
        const blob = Utilities.newBlob(Utilities.base64Decode(String(b.data)), String(b.mime || 'application/pdf'), name);
        const it = DriveApp.getFoldersByName('Кабинет 25-25 · домашка');
        const folder = it.hasNext() ? it.next() : DriveApp.createFolder('Кабинет 25-25 · домашка');
        const file = folder.createFile(blob).setName(st.row[1] + ' — ' + name);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        return {ok: true, url: file.getUrl()};
      }
      case 'hwDone': { need(); const r = setHwDone_(code, b.id, b.text, b.done); return r.ok ? fresh() : r; }
      case 'hwDel': {
        need();
        const sh = ss.getSheetByName('Домашка'), row = Number(b.id);
        if (!(row >= 2 && row <= sh.getLastRow())) return {error: 'no_task'};
        const v = sh.getRange(row, 1, 1, 2).getValues()[0];
        if (String(v[0]).trim() !== code || String(v[1]).trim() !== String(b.text || '').trim()) return {error: 'changed'};
        sh.deleteRow(row);
        const fs = sheet_(ss, 'Файлы');
        if (fs.getLastRow() >= 2) {
          const del = [];
          fs.getRange(2, 1, fs.getLastRow() - 1, 2).getValues().forEach((r, i) => {
            if (String(r[0]).trim() === code && Number(r[1]) === row) del.push(i + 2);
          });
          del.sort((x, y) => y - x).forEach(r => fs.deleteRow(r));
        }
        return fresh();
      }
      case 'gradeAdd': {
        need();
        const p = Number(b.p);
        if (!(p >= 2 && p <= 5)) return {error: 'bad_value'};
        const dv = /^\d{4}-\d{2}-\d{2}$/.test(String(b.d || '')) ? new Date(String(b.d) + 'T00:00:00') : '';
        sheet_(ss, 'Оценки').appendRow([code, dv, String(b.subj || 'математика').trim().slice(0, 60), p, String(b.note || '').trim().slice(0, 200)]);
        return fresh();
      }
      case 'gradeDel': {
        need();
        const sh = sheet_(ss, 'Оценки'), row = Number(b.id);
        if (!(row >= 2 && row <= sh.getLastRow()) || String(sh.getRange(row, 1).getValue()).trim() !== code) return {error: 'changed'};
        sh.deleteRow(row);
        return fresh();
      }
      case 'letterRead': {
        need();
        const sh = sheet_(ss, 'Письма'), row = Number(b.id);
        if (!(row >= 2 && row <= sh.getLastRow()) || String(sh.getRange(row, 1).getValue()).trim() !== code) return {error: 'changed'};
        sh.getRange(row, 4).setValue(true);
        return fresh();
      }
      case 'letterDel': {
        need();
        const sh = sheet_(ss, 'Письма'), row = Number(b.id);
        if (!(row >= 2 && row <= sh.getLastRow()) || String(sh.getRange(row, 1).getValue()).trim() !== code) return {error: 'changed'};
        sh.deleteRow(row);
        return fresh();
      }
      case 'matAdd': {
        need();
        const u = String(b.url || '').trim(), nm = String(b.name || '').trim();
        if (!u || !nm) return {error: 'bad_value'};
        ss.getSheetByName('Разборы').appendRow([code, u, nm, String(b.label || '').trim()]);
        return fresh();
      }
      case 'matDel': {
        need();
        const sh = ss.getSheetByName('Разборы'), row = Number(b.id);
        if (!(row >= 2 && row <= sh.getLastRow()) || String(sh.getRange(row, 1).getValue()).trim() !== code) return {error: 'changed'};
        sh.deleteRow(row);
        return fresh();
      }
      case 'probAdd': {
        const st = need();
        if (progOf_(st.row) !== 'oge9') {
          const p2 = Number(b.p);
          let v = (p2 >= 2 && p2 <= 5) ? p2 : null;

        }

        if (b.p === '' || !(p >= 0 && p <= 31)) return {error: 'bad_value'};

        need();
        const sh = ss.getSheetByName('Пробники'), row = Number(b.id);
        if (!(row >= 2 && row <= sh.getLastRow()) || String(sh.getRange(row, 1).getValue()).trim() !== code) return {error: 'changed'};
        sh.deleteRow(row);

      }

    return {error: 'bad_op'};

function listStudents_() {
  const ss = SpreadsheetApp.getActive(), sh = ss.getSheetByName('Ученики');
  if (!sh || sh.getLastRow() < 2) return [];
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 19).getValues();
  const times = sh.getRange(2, 6, sh.getLastRow() - 1, 1).getDisplayValues();
  const mapSh = ss.getSheetByName('Карта'), maps = {};
  if (mapSh && mapSh.getLastRow() >= 2) mapSh.getRange(2, 1, mapSh.getLastRow() - 1, Math.max(mapSh.getLastColumn(), 2)).getValues().forEach(r => { maps[String(r[0]).trim()] = r; });
  const hwSh = ss.getSheetByName('Домашка'), left = {};
  if (hwSh && hwSh.getLastRow() >= 2) hwSh.getRange(2, 1, hwSh.getLastRow() - 1, 7).getValues().forEach(r => {
    const c = String(r[0]).trim(); if (c && String(r[1]).trim() && r[6] !== true) left[c] = (left[c] || 0) + 1;
  });
  const letters = {};
  const ltSh = ss.getSheetByName('Письма');
  if (ltSh && ltSh.getLastRow() >= 2) ltSh.getRange(2, 1, ltSh.getLastRow() - 1, 4).getValues().forEach(r => {
    const c = String(r[0]).trim(); if (c && String(r[2]).trim() && r[3] !== true) letters[c] = (letters[c] || 0) + 1;
  });
  return rows.map((r, i) => ({r: r, time: String(times[i][0]).trim()})).filter(x => String(x.r[0]).trim()).map(x => {
    const r = x.r, c = String(r[0]).trim(), m = maps[c];
    const pr = progOf_(r), nmax = nMax_(pr);
    let p = 0;
    if (m) for (let n = 1; n <= nmax; n++) if (STATUS_KEY[String(m[n + 1]).trim().toLowerCase()] === 'solid') p += ptsOf_(pr, n);
    return {code: c, name: String(r[1]), fname: String(r[14] || ''), goal: Number(r[3]) || Math.round(goalMax_(pr) * 0.8),
            days: String(r[4] || ''), time: x.time, active: r[11] !== false, prog: pr, pts: p, hwLeft: left[c] || 0,
            letters: letters[c] || 0, cls: Number(r[18]) || clsOfProg_(pr)};
  });
}

function studentData_(code, admin) {
  const st = findStudentRow_(code);
  if (!st) return {error: 'not_found'};
  const r = st.row;
  if (!admin && r[11] === false) return {error: 'inactive'};
  const ss = SpreadsheetApp.getActive();
  const rows = name => {
    const sh = ss.getSheetByName(name);
    if (!sh || sh.getLastRow() < 2) return [];
    return sh.getRange(2, 1, sh.getLastRow() - 1, sh.getLastColumn()).getValues()
      .map((v, i) => ({v: v, row: i + 2})).filter(x => String(x.v[0]).trim() === code);
  };
  const map = {};
  const m = rows('Карта')[0];
  for (let n = 1; n <= 58; n++) map[n] = STATUS_KEY[String(m ? m.v[n + 1] : '').trim().toLowerCase()] || 'new';
  const notes = {};
  rows('Комментарии').forEach(x => {
    const n = parseInt(x.v[1], 10);
    if (n >= 1 && n <= 58) notes[n] = {t: String(x.v[2] || ''), url: String(x.v[3] || ''), label: String(x.v[4] || '')};
  });
  const skills = {};
  rows('Навыки').forEach(x => {
    const id = String(x.v[1]).trim().toUpperCase(), k = STATUS_KEY[String(x.v[2]).trim().toLowerCase()];
    if (SKILL_RE.test(id) && k && k !== 'new') skills[id] = {s: k, d: day_(x.v[3])};
  });
  const hwFiles = {};
  rows('Файлы').forEach(x => {
    const k = Number(x.v[1]);
    if (k >= 2) (hwFiles[k] = hwFiles[k] || []).push({name: String(x.v[2] || 'PDF для печати'), url: String(x.v[3] || ''), id: x.row});
  });
  const hw = rows('Домашка').filter(x => String(x.v[1]).trim()).map(x => ({
    id: x.row, t: String(x.v[1]).trim(), n: String(x.v[2] || ''), due: day_(x.v[3]),
    url: String(x.v[4] || ''), label: String(x.v[5] || ''), done: x.v[6] === true, file: String(x.v[8] || ''),
    sec: HW_SEC_KEY[String(x.v[9] || '').trim().toLowerCase()] || '',
    files: hwFiles[x.row] || (String(x.v[8] || '').trim() ? [{name: 'PDF с заданиями', url: String(x.v[8]).trim()}] : [])
  }));
  const grades = rows('Оценки')
    .map(x => ({id: x.row, d: day_(x.v[1]), subj: String(x.v[2] || '').trim(), p: Number(x.v[3]), note: String(x.v[4] || '').trim()}))
    .filter(g => g.p >= 2 && g.p <= 5)
    .sort((a, b) => String(a.d).localeCompare(String(b.d)));
  const hist = rows('Пробники').filter(x => String(x.v[3]).trim() !== '' && (!isNaN(Number(x.v[3])) || /зач/i.test(String(x.v[3]))))
    .map(x => ({id: x.row, d: day_(x.v[1]), t: String(x.v[2] || ''), p: isNaN(Number(x.v[3])) ? (/^\s*не/i.test(String(x.v[3])) ? 'незачёт' : 'зачёт') : Number(x.v[3])}))
    .sort((a, b) => String(a.d).localeCompare(String(b.d)));
  const mats = rows('Разборы')
    .filter(x => String(x.v[1]).trim() && String(x.v[2]).trim())
    .map(x => [String(x.v[1]).trim(), String(x.v[2]).trim(), String(x.v[3] || '').trim(), x.row]);
  const matsSh = ss.getSheetByName('Разборы');
  const matsAll = (matsSh && matsSh.getLastRow() >= 2) ? matsSh.getLastRow() - 1 : 0;
  const prog = progOf_(r);
  const out = {
    name: String(r[1]), dat: String(r[2] || r[1]), goal: Number(r[3]) || Math.round(goalMax_(prog) * 0.8),
    days: String(r[4] || ''), time: st.time, topic: String(r[6] || ''), board: String(r[7] || ''),
    work: String(r[8] || '').split(';').map(s => s.trim()).filter(Boolean),
    lessonsPlan: Number(r[9]) || 0, lessonsDone: Number(r[10]) || 0,
    map: map, notes: notes, skills: skills, hw: hw, hist: hist, mats: mats, matsAll: matsAll, prog: prog,
    // Без пароля (публичная ссылка ученика) родитель, телефон и год рождения не отдаются:
    // для поздравления хватает дня и месяца, год подменяется на 2000.
    fname: String(r[14] || ''), bdate: admin ? day_(r[15]) : maskBdate_(day_(r[15])),
    parent: admin ? String(r[16] || '') : '', phone: admin ? String(r[17] || '') : '',
    cls: Number(r[18]) || clsOfProg_(prog), grades: grades
  };
  if (admin) {
    out.active = r[11] !== false; out.code = code;
    const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
    out.letters = rows('Письма').map(x => ({
      id: x.row, d: day_(x.v[1]),
      time: x.v[1] instanceof Date ? Utilities.formatDate(x.v[1], tz, 'HH:mm') : '',
      text: String(x.v[2] || ''), read: x.v[3] === true
    })).sort((a, b) => b.id - a.id);
  }
  return out;
}

/* ================= служебное ================= */
function findStudentRow_(code) {
  code = String(code || '').trim();
  if (!code) return null;
  const sh = SpreadsheetApp.getActive().getSheetByName('Ученики');
  if (!sh || sh.getLastRow() < 2) return null;
  const v = sh.getRange(2, 1, sh.getLastRow() - 1, 19).getValues();
  for (let i = 0; i < v.length; i++) if (String(v[i][0]).trim() === code)
    return {row: v[i], index: i + 2, time: String(sh.getRange(i + 2, 6).getDisplayValue()).trim()};
  return null;
}
function findRow_(name, test) {
  const sh = SpreadsheetApp.getActive().getSheetByName(name);
  if (!sh || sh.getLastRow() < 2) return 0;
  const v = sh.getRange(2, 1, sh.getLastRow() - 1, Math.max(2, sh.getLastColumn())).getValues();
  for (let i = 0; i < v.length; i++) if (test(v[i])) return i + 2;
  return 0;
}
function newCode_() {
  const abc = 'abcdefghjkmnpqrstuvwxyz23456789';
  let s = '';
  for (let i = 0; i < 10; i++) s += abc[Math.floor(Math.random() * abc.length)];
  return findStudentRow_(s) ? newCode_() : s;
}
function day_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, SpreadsheetApp.getActive().getSpreadsheetTimeZone(), 'yyyy-MM-dd');
  const m = String(v || '').match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})$/);
  return m ? m[3] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[1]).slice(-2) : '';
}
function maskBdate_(iso) {
  const m = String(iso || '').match(/^\d{4}-(\d{2}-\d{2})$/);
  return m ? '2000-' + m[1] : '';
}
function ruDate_(iso) {
  const m = String(iso || '').match(/^(\d{4})-(\d{2})-(\d{2})$/);
  return m ? m[3] + '.' + m[2] + '.' + m[1] : '';
}
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
